package com.lidao.moran.worldgen;

import net.minecraft.block.Blocks;
import net.minecraft.util.Identifier;
import net.minecraft.util.registry.Registry;
import net.minecraft.world.gen.surfacebuilder.MaterialRules;
import terrablender.api.SurfaceRuleManager;

public class SurfaceRuleRegistry {
    public static void initialize() {
        // 规则优先级：从上到下，先匹配先执行
        SurfaceRuleManager.addSurfaceRule(
            SurfaceRuleManager.RuleCategory.OVERWORLD, // 或自定义维度类型
            "mo-mod",  // 模组命名空间
            MaterialRules.sequence(
                // 规则 1：桃花群系的水下底部 → 桃花沙
                MaterialRules.condition(
                    MaterialRules.and(
                        MaterialRules.biome(RegistryKey.of(RegistryKeys.BIOME, new Identifier("mo-mod", "peach_blossom_biome"))),
                        MaterialRules.waterBlockCheck(-1, 0),  // 水下 1 格
                        MaterialRules.stoneDepthCheck(0, false, 0, MaterialRules.StoneDepthType.FLOOR)
                    ),
                    MaterialRules.state(Registry.BLOCK.get(new Identifier("mo-mod", "peach_sand")).getDefaultState())
                ),
                
                // 规则 2：桃花群系的地表 → 桃花土
                MaterialRules.condition(
                    MaterialRules.biome(RegistryKey.of(RegistryKeys.BIOME, new Identifier("mo-mod", "peach_blossom_biome"))),
                    MaterialRules.sequence(
                        // 地表第 1 层（草）
                        MaterialRules.condition(
                            MaterialRules.stoneDepthCheck(1, false, 0, MaterialRules.StoneDepthType.FLOOR),
                            MaterialRules.state(Registry.BLOCK.get(new Identifier("mo-mod", "peach_grass")).getDefaultState())
                        ),
                        // 地表下方（泥土）
                        MaterialRules.condition(
                            MaterialRules.stoneDepthCheck(3, false, 0, MaterialRules.StoneDepthType.FLOOR),
                            MaterialRules.state(Registry.BLOCK.get(new Identifier("mo-mod", "peach_dirt")).getDefaultState())
                        ),
                        // 更深处（石头）
                        MaterialRules.state(Registry.BLOCK.get(new Identifier("mo-mod", "peach_stone")).getDefaultState())
                    )
                ),
                
                // 规则 3：其他群系 → 原版（fallback）
                MaterialRules.condition(
                    MaterialRules.not(MaterialRules.biome(RegistryKey.of(RegistryKeys.BIOME, new Identifier("mo-mod", "peach_blossom_biome")))),
                    MaterialRules.state(Blocks.STONE.getDefaultState())
                )
            )
        );
    }
}
